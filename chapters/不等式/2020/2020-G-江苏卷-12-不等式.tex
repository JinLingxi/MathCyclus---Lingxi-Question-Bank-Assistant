\begin{problem}{2020}{G}{江苏卷}{12}{不等式}
已知 $ 5x^2y^2 + y^4 = 1 $ ($ x, y \in \mathbb{R} $)，则 $ x^2 + y^2 $ 的最小值是 \underline{\hspace{4em}}.
\end{problem}

\begin{answer}
$ \dfrac{4}{5} $
\end{answer}

\begin{solutions}
显然 $ y \ne 0 $，$ x^2 = \dfrac{1 - y^4}{5y^2} = \dfrac{1}{5y^2} - \dfrac{y^2}{5} $  
$ x^2 + y^2 = \dfrac{1}{5y^2} + \dfrac{4y^2}{5} \ge 2\sqrt{ \dfrac{4}{25} } = \dfrac{4}{5} $，取等条件为 $ y^2 = \dfrac{1}{2} $，因此 $ x^2 + y^2 $ 最小值是 $ \dfrac{4}{5} $.
\end{solutions}
