\begin{problem}{2012}{G}{福建卷（理）}{1}{复数}
若复数 $ z $ 满足 $ zi=1-i $，则 $ z $ 等于
\begin{choices}
\choice{{$-1-i$}}
\choice{{$1-i$}}
\choice{{$-1+i$}}
\choice{{$1+i$}}
\end{choices}
\end{problem}

\begin{answer}
(A)
\end{answer}

\begin{solutions}
由 $ zi = 1 - i $，得 $ z = \dfrac{1 - i}{i} = \dfrac{(1 - i)(-i)}{i(-i)} = \dfrac{-i + i^2}{1} = -i - 1 = -1 - i $.
\end{solutions}
