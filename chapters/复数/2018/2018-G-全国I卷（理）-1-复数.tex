\begin{problem}{2018}{G}{全国I卷（理）}{1}{复数}
设 $ z = \dfrac{1 - i}{1 + i} + 2i $，则 $ |z| = $ (\hspace{1cm})
\begin{choices}
\choice{{$0$}}
\choice{{$\dfrac{1}{2}$}}
\choice{{$1$}}
\choice{{$\sqrt{2}$}}
\end{choices}
\end{problem}

\begin{answer}
(C) $1$
\end{answer}

\begin{solutions}
$$
z = \dfrac{1 - i}{1 + i} + 2i = \dfrac{(1 - i)^2}{(1 + i)(1 - i)} + 2i = \dfrac{1 - 2i + i^2}{1 - i^2} + 2i = \dfrac{1 - 2i - 1}{1 + 1} + 2i = \dfrac{-2i}{2} + 2i = -i + 2i = i
$$
所以 $ |z| = |i| = 1 $.
\end{solutions}
