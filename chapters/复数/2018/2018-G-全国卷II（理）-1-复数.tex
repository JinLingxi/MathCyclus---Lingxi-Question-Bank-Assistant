\begin{problem}{2018}{G}{全国卷II（理）}{1}{复数}
$ \dfrac{1+2\mathrm{i}}{1-2\mathrm{i}} = $ (\hspace{1cm})
\begin{choices}
\choice{{$-\dfrac{4}{5}-\dfrac{3}{5}\mathrm{i}$}}
\choice{{$-\dfrac{4}{5}+\dfrac{3}{5}\mathrm{i}$}}
\choice{{$-\dfrac{3}{5}-\dfrac{4}{5}\mathrm{i}$}}
\choice{{$-\dfrac{3}{5}+\dfrac{4}{5}\mathrm{i}$}}
\end{choices}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solutions}
$$
\dfrac{1+2\mathrm{i}}{1-2\mathrm{i}} = \dfrac{(1+2\mathrm{i})^2}{(1-2\mathrm{i})(1+2\mathrm{i})} = \dfrac{1+4\mathrm{i}+4\mathrm{i}^2}{1-4\mathrm{i}^2} = \dfrac{1+4\mathrm{i}-4}{1+4} = \dfrac{-3+4\mathrm{i}}{5} = -\dfrac{3}{5}+\dfrac{4}{5}\mathrm{i}.
$$
\end{solutions}
