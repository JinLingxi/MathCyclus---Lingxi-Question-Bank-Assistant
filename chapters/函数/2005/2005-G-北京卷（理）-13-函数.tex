\begin{problem}{2005}{G}{北京卷（理）}{13}{函数}

\begin{enumerate}
\item $ f(x_1+x_2)=f(x_1)\cdot f(x_2) $；
\item $ f(x_1\cdot x_2)=f(x_1)+f(x_2) $；
\item $ \dfrac{f(x_1)-f(x_2)}{x_1-x_2}>0 $；
\item $ f\left(\dfrac{x_1+x_2}{2}\right)<\dfrac{f(x_1)+f(x_2)}{2} $．
\end{enumerate}
