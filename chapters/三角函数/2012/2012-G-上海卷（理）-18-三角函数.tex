\begin{problem}{2012}{G}{上海卷（理）}{18}{三角函数}
设 $ a_n = \dfrac{1}{n} \sin \dfrac{n\pi}{25} $，$ S_n = a_1 + a_2 + \cdots + a_n $，在 $ S_1, S_2, \cdots, S_{100} $ 中，正数的个数是（\hspace{1cm}） (\hspace{1cm})
\begin{choices}
\choice{{$25$}}
\choice{{$50$}}
\choice{{$75$}}
\choice{{$100$}}
\end{choices}
\end{problem}

\begin{answer}
$ D $
\end{answer}

\begin{solutions}
$ a_1, a_2, \cdots, a_{24} > 0 $，$ a_{25} = 0 $，因此 $ S_1, S_2, \cdots, S_{25} > 0 $；

$ a_{26}, a_{27}, \cdots, a_{49} < 0 $，且 $ a_{26} = \dfrac{\sin \dfrac{26}{25}\pi}{26} = -\dfrac{\sin \dfrac{\pi}{25}}{26} $，$ a_1 + a_{26} = \dfrac{25}{26} \sin \dfrac{\pi}{25} > 0 $，同理 $ a_2 + a_{27} > 0 $，$ \cdots $，$ a_{24} + a_{49} > 0 $，$ a_{25} + a_{50} = 0 $，进而 $ S_{26}, S_{27}, \cdots, S_{50} > 0 $；

$ a_{51}, a_{52}, \cdots, a_{74} > 0 $，$ a_{75} = 0 $，因此 $ S_{51}, S_{52}, \cdots, S_{75} > 0 $；

$ a_{76}, a_{77}, \cdots, a_{99} < 0 $，$ a_{100} = 0 $，且 $ a_{76} = \dfrac{\sin \dfrac{76}{25}\pi}{76} = -\dfrac{\sin \dfrac{\pi}{25}}{76} $，

$ a_{51} + a_{76} = \left( \dfrac{1}{51} - \dfrac{1}{76} \right) \sin \dfrac{\pi}{25} > 0 $，同理 $ a_{52} + a_{77} > 0 $，$ \cdots $，$ a_{74} + a_{99} > 0 $，$ a_{75} + a_{100} = 0 $，

进而 $ S_{76}, S_{77}, \cdots, S_{100} > 0 $．

综上选 $ D $．
\end{solutions}
