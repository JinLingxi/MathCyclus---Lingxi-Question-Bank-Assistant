\begin{problem}{2008}{G}{上海卷}{12}{排列组合}
组合数 $ C_n^r $（$ n>r\geq 1 $，$ n $、$ r\in\mathbb{Z} $）恒等于 (\hspace{1cm})
\begin{choices}
\choice{{$\dfrac{r+1}{n+1}C_{n-1}^{r-1}$}}
\choice{{$(n+1)(r+1)C_{n-1}^{r-1}$}}
\choice{{$nrC_{n-1}^{r-1}$}}
\choice{{$\dfrac{n}{r}C_{n-1}^{r-1}$}}
\end{choices}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solutions}
由组合数公式可知 $ C_n^r=\dfrac{n!}{r!(n-r)!} $，$ C_{n-1}^{r-1}=\dfrac{(n-1)!}{(r-1)!(n-r)!} $，由此应选 D.
\end{solutions}
