\begin{choices}
\choice{{若 $ n=1 $，则 $ H(X)=0 $}}
\choice{{若 $ n=2 $，则 $ H(X) $ 随着 $ p_1 $ 的增大而增大}}
\choice{{若 $ \displaystyle p_i=\dfrac{1}{n} $（$ i=1 $，$ 2 $，$ \cdots $，$ n $），则 $ H(X) $ 随着 $ n $ 的增大而增大}}
\choice{{若 $ n=2m $，随机变量 $ Y $ 所有可能的取值为 $ 1 $，$ 2 $，$ \cdots $，$ m $，且 $ P(Y=j)=p_j+p_{2m+1-j} $（$ j=1 $，$ 2 $，$ \cdots $，$ m $），则 $ H(X)\leqslant H(Y) $}}
\end{choices}
