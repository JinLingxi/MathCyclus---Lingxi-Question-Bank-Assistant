\begin{problem}{2006}{G}{江西卷}{8}{二项式定理}
在 $(x-\sqrt{2})^{2006}$ 的二项展开式中，含 $x$ 的奇次幂的项之和为 $S$，当 $x=\sqrt{2}$ 时，$S$ 等于 (\hspace{1cm})
\begin{choices}
\choice{{$2^{3008}$}}
\choice{{$-2^{3008}$}}
\choice{{$2^{3009}$}}
\choice{{$-2^{3009}$}}
\end{choices}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solutions}
$(x-\sqrt{2})^{2006}=a_0x^{2006}+a_1x^{2005}+\cdots+a_{2005}x+a_{2006}$．令 $x=\sqrt{2}$ 与 $x=-\sqrt{2}$，可得

$$
0=a_0(\sqrt{2})^{2006}+a_1(\sqrt{2})^{2005}+\cdots+a_{2005}\sqrt{2}+a_{2006},
$$
$$
2^{3009}=a_0(\sqrt{2})^{2006}-a_1(\sqrt{2})^{2005}+\cdots-a_{2005}\sqrt{2}+a_{2006},
$$

两式相减除以 $2$，可得 $a_1(\sqrt{2})^{2005}+a_3(\sqrt{2})^{2003}+\cdots+a_{2005}\sqrt{2}=-2^{3009}\div 2=-2^{3008}$．
\end{solutions}
