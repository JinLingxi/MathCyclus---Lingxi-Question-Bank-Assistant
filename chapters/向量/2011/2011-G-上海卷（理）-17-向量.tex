\begin{problem}{2011}{G}{上海卷（理）}{17}{向量}
设 $ A_1 $, $ A_2 $, $ A_3 $, $ A_4 $, $ A_5 $ 是平面上给定的 5 个不同点，则使 $ \overrightarrow{MA_1} + \overrightarrow{MA_2} + \overrightarrow{MA_3} + \overrightarrow{MA_4} + \overrightarrow{MA_5} = \vec{0} $ 成立的点 $ M $ 的个数为 (\hspace{1cm})
\begin{choices}
\choice{{$0$}}
\choice{{$1$}}
\choice{{$5$}}
\choice{{$10$}}
\end{choices}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solutions}
由 $ \overrightarrow{MA_1} + \overrightarrow{MA_2} + \overrightarrow{MA_3} + \overrightarrow{MA_4} + \overrightarrow{MA_5} = \vec{0} $，得  
$$
\sum_{i=1}^{5} (\vec{A_i} - \vec{M}) = \vec{0} \quad \Rightarrow \quad \sum_{i=1}^{5} \vec{A_i} - 5\vec{M} = \vec{0} \quad \Rightarrow \quad \vec{M} = \frac{1}{5} \sum_{i=1}^{5} \vec{A_i}.
$$  
即 $ M $ 为五点的重心，唯一存在。  
\end{solutions}
