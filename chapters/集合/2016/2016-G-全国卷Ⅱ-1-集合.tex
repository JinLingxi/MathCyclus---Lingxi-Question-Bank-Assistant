\begin{problem}{2016}{G}{全国卷Ⅱ}{1}{集合}
1. 已知集合 $ A = \{1,2,3\} $，$ B = \{x \mid x^2 < 9\} $，则 $ A \cap B = $ (\hspace{1cm})
\begin{choices}
\choice{{$\{-2,-1,0,1,2,3\}$}}
\choice{{$\{-2,-1,0,1,2\}$}}
\choice{{$\{1,2,3\}$}}
\choice{{$\{1,2\}$}}
\end{choices}
\end{problem}

\begin{answer}
(D) $\{1,2\}$
\end{answer}

\begin{solutions}
由 $ x^2 < 9 $ 得 $ -3 < x < 3 $，故 $ B = \{-2,-1,0,1,2\} $，又 $ A = \{1,2,3\} $，所以 $ A \cap B = \{1,2\} $.
\end{solutions}
