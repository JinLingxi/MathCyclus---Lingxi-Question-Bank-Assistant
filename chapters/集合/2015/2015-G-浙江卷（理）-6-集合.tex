\begin{problem}{2015}{G}{浙江卷（理）}{6}{集合}
设 $ A $, $ B $ 是有限集，定义：$ d(A,B)=\operatorname{card}(A\cup B)-\operatorname{card}(A\cap B) $，其中 $ \operatorname{card}(A) $ 表示有限集 $ A $ 中元素的个数。  
命题\circled{1}：对任意有限集 $ A $, $ B $，“$ A\neq B $”是“$ d(A,B)>0 $”的充分必要条件；  
命题\circled{2}：对任意有限集 $ A $, $ B $, $ C $，$ d(A,C)\leq d(A,B)+d(B,C) $.  
(\hspace{1cm})  
\begin{choices}
\choice{{命题\circled{1}和命题\circled{2}都成立}}
\choice{{命题\circled{1}和命题\circled{2}都不成立}}
\choice{{命题\circled{1}成立，命题\circled{2}不成立}}
\choice{{命题\circled{1}不成立，命题\circled{2}成立}}
\end{choices}
\end{problem}
