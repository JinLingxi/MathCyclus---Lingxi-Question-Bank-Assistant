\begin{problem}{2025}{G}{全国II卷}{3}{集合}
已知集合 $A=\{-4,0,1,2,8\}$，$B=\{x \mid x^3=x\}$，则 $A \cap B =$
(\hspace{1cm})
\begin{choices}
\choice{{$\{0,1,2\}$}}
\choice{{$\{1,2,8\}$}}
\choice{{$\{2,8\}$}}
\choice{{$\{0,1\}$}}
\end{choices}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solutions}
由 $x^3 = x$ 可得 $x^3 - x = 0$.

因式分解得 $x(x-1)(x+1) = 0$.

解得 $x = 0$ 或 $x = 1$ 或 $x = -1$.

所以 $B = \{-1, 0, 1\}$.

已知 $A = \{-4, 0, 1, 2, 8\}$.

取公共元素得 $A \cap B = \{0, 1\}$.

故选 \boxed{D}.
\end{solutions}
