\begin{problem}{2013}{G}{广东卷（理）}{8}{集合}

\begin{choices}
\choice{{$(y,z,w)\in S$，$(x,y,w)\notin S$}}

\choice{{$(y,z,w)\notin S$，$(x,y,w)\in S$}}
\choice{{$(y,z,w)\notin S$，$(x,y,w)\notin S$}}
\end{choices}
