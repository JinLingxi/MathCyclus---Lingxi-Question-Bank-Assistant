\begin{problem}{2019}{G}{浙江卷}{7}{集合}
设 $0<a<1$，则随机变量 $X$ 的分布列是：
\begin{tabular}{|c|ccc|}
\hline
$X$ & $0$ & $a$ & $1$ \\
\hline
$P$ & $\dfrac{1}{3}$ & $\dfrac{1}{3}$ & $\dfrac{1}{3}$\\
\hline
\end{tabular}

则当 $a$ 在 $(0,1)$ 内增大时 (\hspace{1cm})
\begin{choices}
\choice{{$D(X)$ 增大}}
\choice{{$D(X)$ 减小}}
\choice{{$D(X)$ 先增大后减小}}
\choice{{$D(X)$ 先减小后增大}}
\end{choices}
\end{problem}

\begin{answer}
$D$
\end{answer}

\begin{solutions}
$$
E(X)=0\times\frac{1}{3}+a\times\frac{1}{3}+1\times\frac{1}{3}=\frac{a+1}{3}
$$

$$
D(X)=\left(\frac{a+1}{3}\right)^{2}\times\frac{1}{3}+\left(a-\frac{a+1}{3}\right)^{2}\times\frac{1}{3}+\left(1-\frac{a+1}{3}\right)^{2}\times\frac{1}{3}
$$

$$
=\frac{1}{27}\left[(a+1)^{2}+(2a-1)^{2}+(a-2)^{2}\right]=\frac{2}{9}(a^{2}-a+1)=\frac{2}{9}\left(a-\frac{1}{2}\right)^{2}+\frac{1}{6}
$$

因为 $0<a<1$, 所以 $D(X)$ 先减小后增大，故选 $D$.
\end{solutions}
