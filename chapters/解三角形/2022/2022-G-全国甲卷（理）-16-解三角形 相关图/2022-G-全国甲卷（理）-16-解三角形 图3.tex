\begin{tikzpicture}[line cap=round,line join=round,scale=0.6]

\pgfmathsetmacro{\R}{sqrt(10)}

\coordinate (B) at (-3,0);
\coordinate (C) at (3,0);
\coordinate (D) at (-1,0);

\draw[thick] (A)--(D);

\draw[densely dashed] (B)--(E)--(C);
\draw[densely dashed] (D)--(E);

\node[left] at (O) {$O$};
